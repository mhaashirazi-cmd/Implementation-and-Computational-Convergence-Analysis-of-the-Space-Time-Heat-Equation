// Exact solution 
template <int dim>
class ExactSolution : public Function<dim>
{
public:
  ExactSolution() : Function<dim>() {}
  virtual double value(const Point<dim> &p, const unsigned int /*component*/) const override
{
  const double pi = numbers::PI;
  const double x = p[0];
  const double y = p[1];
  const double t = p[2]; // Time is now the 3rd dimension

  // Peak center moves in a circle
  const double x0 = 0.5 + 0.25 * std::cos(2.0 * pi * t);
  const double y0 = 0.5 + 0.25 * std::sin(2.0 * pi * t);
  
  const double r2 = (x - x0) * (x - x0) + (y - y0) * (y - y0);
  return 1.0 / (1.0 + r2);
}
};

class Step4
{
public:
  Step4();
  void run();

private:
  void make_grid(const unsigned int cycle);
  void setup_system();
  void assemble_system();
  void solve();
  void output_results(const unsigned int cycle) const;

  Triangulation<3>     triangulation; // 3D for 2D space + time [cite: 14]
  FE_Q<3>              fe;
  DoFHandler<3>        dof_handler;
  SparsityPattern      sparsity_pattern;
  SparseMatrix<double> system_matrix;
  Vector<double>       solution;
  Vector<double>       system_rhs;
};

Step4::Step4() : fe(1), dof_handler(triangulation) {}

void Step4::make_grid(const unsigned int cycle)
{
  triangulation.clear();
  // Space-time domain: x in [0,1], t in [0,1] [cite: 11]
  GridGenerator::hyper_cube(triangulation, 0.0, 1.0);
  triangulation.refine_global(cycle);
}

void Step4::setup_system()
{
  dof_handler.distribute_dofs(fe);
  DynamicSparsityPattern dsp(dof_handler.n_dofs());
  DoFTools::make_sparsity_pattern(dof_handler, dsp);
  sparsity_pattern.copy_from(dsp);
  system_matrix.reinit(sparsity_pattern);
  solution.reinit(dof_handler.n_dofs());
  system_rhs.reinit(dof_handler.n_dofs());
}

void Step4::assemble_system()
{
  QGauss<3> quadrature_formula(fe.degree + 1);
  QGauss<2> face_quadrature_formula(fe.degree + 1); // For initial condition [cite: 112]

  FEValues<3> fe_values(fe, quadrature_formula,
                        update_values | update_gradients | 
                        update_quadrature_points | update_JxW_values);

  FEFaceValues<3> fe_face_values(fe, face_quadrature_formula,
                                 update_values | update_quadrature_points | 
                                 update_JxW_values);

 const unsigned int dofs_per_cell = fe.n_dofs_per_cell();
  FullMatrix<double> cell_matrix(dofs_per_cell, dofs_per_cell);
  Vector<double>     cell_rhs(dofs_per_cell);
  std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);

  for (const auto &cell : dof_handler.active_cell_iterators())
    {
      fe_values.reinit(cell);
      cell_matrix = 0;
      cell_rhs    = 0;

      for (const unsigned int q_index : fe_values.quadrature_point_indices())
        {
          const Point<3> &p = fe_values.quadrature_point(q_index);
          const double x = p[0];
          const double y = p[1];
          const double t = p[2];
          const double pi = numbers::PI;

          const double x0 = 0.5 + 0.25 * std::cos(2.0 * pi * t);
            const double y0 = 0.5 + 0.25 * std::sin(2.0 * pi * t);
            const double x0_dot = -0.5 * pi * std::sin(2.0 * pi * t);
            const double y0_dot = 0.5 * pi * std::cos(2.0 * pi * t);

            const double r2 = (x - x0) * (x - x0) + (y - y0) * (y - y0);
            const double den = 1.0 + r2;

            const double du_dt = (2.0 * (x - x0) * x0_dot + 2.0 * (y - y0) * y0_dot) / (den * den);
            const double laplacian_u = (8.0 * r2) / (den * den * den) - 4.0 / (den * den);
            const double forcing_f = du_dt - laplacian_u;
          // Exact forcing term f(t,x) [cite: 16, 265, 301]

for (const unsigned int i : fe_values.dof_indices()) {
                for (const unsigned int j : fe_values.dof_indices()) {
                    // Update matrix with 3D gradients (dx, dy, dt)
                    cell_matrix(i, j) += (fe_values.shape_grad(j, q_index)[2] * fe_values.shape_value(i, q_index) + // dt
                                          fe_values.shape_grad(j, q_index)[0] * fe_values.shape_grad(i, q_index)[0] + // dx
                                          fe_values.shape_grad(j, q_index)[1] * fe_values.shape_grad(i, q_index)[1]) * // dy
                                         fe_values.JxW(q_index);
                }
              cell_rhs(i) += fe_values.shape_value(i, q_index) * forcing_f * fe_values.JxW(q_index);
            }
        }

      // Enforcement of Initial Condition u(0,x) = sin(pi*x) [cite: 266, 267]
      for (unsigned int face_n = 0; face_n < GeometryInfo<2>::faces_per_cell; ++face_n)
        if (cell->at_boundary(face_n) && std::abs(cell->face(face_n)->center()[2] - 0.0) < 1e-10)
          {
            fe_face_values.reinit(cell, face_n);
            for (const unsigned int q_f : fe_face_values.quadrature_point_indices())
              {
                const double x_f = fe_face_values.quadrature_point(q_f)[0];
const double y_f = fe_face_values.quadrature_point(q_f)[1];
const double r2_f = (x_f - 0.75) * (x_f - 0.75) + (y_f - 0.5) * (y_f - 0.5); // t=0 peaks at x=0.75, y=0.5
const double u0_val = 1.0 / (1.0 + r2_f);
                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  cell_rhs(i) += u0_val * fe_face_values.shape_value(i, q_f) * fe_face_values.JxW(q_f);
              }
          }

      cell->get_dof_indices(local_dof_indices);
      for (unsigned int i = 0; i < dofs_per_cell; ++i)
        {
          for (unsigned int j = 0; j < dofs_per_cell; ++j)
            system_matrix.add(local_dof_indices[i], local_dof_indices[j], cell_matrix(i, j));
          system_rhs(local_dof_indices[i]) += cell_rhs(i);
        }
    }

  // Dirichlet Boundary Conditions u=0 at x=0 and x=1 [cite: 268]
  std::map<types::global_dof_index, double> boundary_values;
  VectorTools::interpolate_boundary_values(dof_handler, 0, ExactSolution<3>(), boundary_values);
  MatrixTools::apply_boundary_values(boundary_values, system_matrix, solution, system_rhs);
}

void Step4::solve()
{
  SolverControl solver_control(10000, 1e-10);
  
  // Update: Add AdditionalData(50) to allow GMRES to restart every 50 iterations
  SolverGMRES<Vector<double>> solver(solver_control,
                                     SolverGMRES<Vector<double>>::AdditionalData(50));

  PreconditionJacobi<SparseMatrix<double>> preconditioner;
  preconditioner.initialize(system_matrix);

  solver.solve(system_matrix, solution, system_rhs, preconditioner);
}

void Step4::output_results(const unsigned int cycle) const
{
  DataOut<3> data_out;
  data_out.attach_dof_handler(dof_handler);
  data_out.add_data_vector(solution, "solution");
  data_out.build_patches();
  std::ofstream output("solution-" + std::to_string(cycle) + ".vtk");
  data_out.write_vtk(output);
}

void Step4::run()
{
  for (unsigned int cycle = 0; cycle < 6; ++cycle) // Refinement loop for convergence analysis [cite: 295, 319]
    {
      std::cout << "Cycle " << cycle << ":" << std::endl;
      make_grid(cycle);
      setup_system();
      assemble_system();
      solve();

      Vector<double> difference_per_cell(triangulation.n_active_cells());
      VectorTools::integrate_difference(dof_handler, solution, ExactSolution<3>(),
                                        difference_per_cell, QGauss<3>(fe.degree + 1),
                                        VectorTools::L2_norm);
      const double l2_error = VectorTools::compute_global_error(triangulation, 
                                                               difference_per_cell, 
                                                               VectorTools::L2_norm);

      std::cout << "  Active cells: " << triangulation.n_active_cells() << std::endl
                << "  L2 Error:     " << l2_error << std::endl;
      
      output_results(cycle);
    }
}
